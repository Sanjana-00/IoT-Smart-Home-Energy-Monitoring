/* =====================================================
   NUMBER FUNCTION
   ===================================================== */

function num(v, fallback=0){

  const n =
    Number(v);

  return Number.isFinite(n)
    ? n
    : fallback;

}



/* =====================================================
   UPDATE LIVE DATA
   ===================================================== */

function updateLive(d){

  if(!d)
    return;



  const voltage =
    num(d.voltage);



  const currentMA =

    d.current_ma !== undefined

      ? num(d.current_ma)

      : num(d.current) * 1000;



  const power =
    num(d.power_w);



  const energyKwh =
    num(d.energy_kwh);



  const energyWh =
    energyKwh * 1000;



  /* DISPLAY VOLTAGE */

  document.getElementById("voltage")
    .textContent =
    voltage.toFixed(1);



  /* DISPLAY CURRENT */

  document.getElementById("current")
    .textContent =
    Math.round(currentMA);



  /* DISPLAY POWER */

  document.getElementById("power")
    .textContent =
    power.toFixed(1);



  /* DISPLAY ENERGY */

  document.getElementById("energyWh")
    .textContent =
    energyWh.toFixed(2);



  document.getElementById("energyKwh")
    .textContent =
    energyKwh.toFixed(5);



  /* RELAY */

  if(typeof d.relay === "boolean"){

    updateRelay(d.relay);

  }



  /* GRAPH TIME */

  const time =
    new Date().toLocaleTimeString();



  labels.push(time);

  voltageData.push(voltage);

  currentData.push(currentMA);



  energyChart.data.labels.push(time);

  energyData.push(energyWh);

  powerData.push(power);



  const maxPoints = 60;



  while(labels.length > maxPoints){

    labels.shift();

    voltageData.shift();

    currentData.shift();

  }



  while(
    energyChart.data.labels.length
    > maxPoints
  ){

    energyChart.data.labels.shift();

    energyData.shift();

    powerData.shift();

  }



  liveChart.update("none");

  energyChart.update("none");

}

/* =====================================================
   LINEAR REGRESSION
   ===================================================== */

/*
   Reads power history from:

       /device/history

   and calculates:

       y = mx + b

   using the least-squares method.
*/


function calculateRegression(history){

  if(
    !history ||
    typeof history !== "object"
  ){

    return{

      nextPower:0,

      forecast30:0,

      points:0

    };

  }



  const rows = [];



  Object.keys(history).forEach(key=>{

    const item =
      history[key];


    if(
      !item ||
      typeof item !== "object"
    ){

      return;

    }



    const power =
      Number(item.power_w);



    if(!Number.isFinite(power)){

      return;

    }



    /*
       Try timestamp fields.

       If none exists, Firebase
       key is used.
    */

    let timestamp =
      Number(

        item.timestamp ??
        item.time ??
        item.ts ??
        key

      );



    if(!Number.isFinite(timestamp)){

      timestamp =
        rows.length;

    }



    rows.push({

      timestamp:timestamp,

      power:Math.max(
        0,
        power
      )

    });

  });



  /* Sort oldest → newest */

  rows.sort(
    (a,b)=>
      a.timestamp -
      b.timestamp
  );



  /*
     Use recent 100 readings.
  */

  const recent =
    rows.slice(-100);



  if(recent.length < 2){

    return{

      nextPower:0,

      forecast30:0,

      points:recent.length

    };

  }



  /*
     Convert timestamps into X values.
  */

  const firstTime =
    recent[0].timestamp;



  const points =
    recent.map(
      (item,index)=>{

        let timeHours;



        /*
           Millisecond timestamp.
        */

        if(
          Math.abs(
            item.timestamp
          ) > 10000000000
        ){

          timeHours =

            (
              item.timestamp -
              firstTime
            ) / 3600000;

        }

        else{

          /*
             If timestamp format
             is unknown, use point
             sequence.
          */

          timeHours =
            index;

        }



        return{

          x:timeHours,

          y:item.power

        };

      }
    );



  /* =================================================
     LEAST SQUARES
     ================================================= */


  const n =
    points.length;


  let sumX = 0;

  let sumY = 0;

  let sumXY = 0;

  let sumX2 = 0;



  points.forEach(point=>{

    sumX +=
      point.x;

    sumY +=
      point.y;

    sumXY +=
      point.x *
      point.y;

    sumX2 +=
      point.x *
      point.x;

  });



  const denominator =

    n * sumX2 -
    sumX * sumX;



  let slope = 0;

  let intercept =
    sumY / n;



  if(
    Math.abs(
      denominator
    ) > 0.000000001
  ){

    slope =

      (
        n * sumXY -
        sumX * sumY
      ) /
      denominator;



    intercept =

      (
        sumY -
        slope * sumX
      ) /
      n;

  }



  /*
     Predict next power.
  */

  const lastX =
    points[
      points.length - 1
    ].x;



  let nextPower =

    intercept +
    slope *
    (lastX + 1);



  /*
     Don't allow negative power.
  */

  nextPower =
    Math.max(
      0,
      nextPower
    );



  const latestPower =

    points[
      points.length - 1
    ].y;



  /*
     If prediction is invalid,
     use latest power.
  */

  if(
    !Number.isFinite(
      nextPower
    )
  ){

    nextPower =
      latestPower;

  }



  /* =================================================
     30-DAY FORECAST
     ================================================= */

  let forecastWh =
    0;



  /*
     30 days × 24 hours
     = 720 hours
  */

  for(
    let hour=1;
    hour<=720;
    hour++
  ){

    let predicted =

      intercept +
      slope *
      (lastX + hour);



    predicted =
      Math.max(
        0,
        predicted
      );



    forecastWh +=
      predicted;

  }



  const forecast30Kwh =
    forecastWh / 1000;



  return{

    nextPower:
      nextPower,

    forecast30:
      forecast30Kwh,

    points:
      recent.length

  };

}

}
